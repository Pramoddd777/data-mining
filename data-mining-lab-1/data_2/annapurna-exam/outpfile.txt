HISTORICAL PRODUCT JOIN
-----------------------

Query:

CREATE OR REPLACE VIEW sales_enriched AS
SELECT
    s.bill_no,
    s.line_no,
    s.store_id,
    s.product_code,
    p.product_sk,
    p.product_name,
    p.category_id,
    s.business_date,
    s.ts,
    s.qty,
    s.unit_price,
    s.line_type,
    s.revenue_amount
FROM sales_final s
LEFT JOIN pg.public.products p
    ON s.product_code = p.product_code
   AND s.business_date >= p.valid_from
   AND (
        p.valid_to IS NULL
        OR s.business_date <= p.valid_to
   );


Row count:

SELECT COUNT(*) AS rows
FROM sales_enriched;

STORE/MONTH DASHBOARD
---------------------

CREATE OR REPLACE TABLE dashboard_store_month AS
SELECT
    store_id,
    year(business_date) AS sale_year,
    month(business_date) AS sale_month,
    SUM(revenue_amount) AS revenue
FROM sales_enriched
WHERE line_type IN ('SALE', 'RETURN', 'DISCOUNT', 'VOID')
GROUP BY

    month(business_date)

MARCH HISTORICAL PRICE QUERY
----------------------------

SELECT
    s.product_sk,
    s.product_name,
    pr.selling_price AS march_price,
    SUM(s.qty) AS quantity_sold,
    SUM(s.qty * pr.selling_price) AS march_value
FROM sales_enriched s
JOIN pg.public.price_revisions pr
    ON s.product_sk = pr.product_sk
   AND s.business_date >= pr.effective_from
   AND s.business_date <= pr.effective_to
WHERE s.business_date >= DATE '2024-03-01'
  AND s.business_date < DATE '2024-04-01'
  AND s.line_type IN ('SALE', 'RETURN', 'VOID')
GROUP BY
    s.product_sk,
    s.product_name,
    pr.selling_price
ORDER BY
    s.product_sk
LIMIT 20;

Purpose:

Use product_sk + effective date range to retrieve the price valid during March 2024.

    f.revenue_inr AS finance_revenue,
    ROUND(CAST(d.revenue AS DOUBLE), 2) AS pipeline_revenue,
